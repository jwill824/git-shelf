# git-cloak Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build and test the complete git-cloak tool — a Homebrew-installable zsh CLI for managing personal file overlays (skip-worktree + scoped gitignore) safely across git branch operations.

**Architecture:** Single dispatcher at `bin/git-cloak` routes to `lib/cmd-*.sh` scripts via `source`, sharing path resolution and helpers from `lib/common.sh`. Three config scopes (global/workspace/project) are composed into a single `.git/personal/ignore` file per repo. Safe branch operations (switch/refresh/merge-main) follow a stash-operate-pop lifecycle with configurable conflict handling via a per-project monitor list.

**Tech Stack:** zsh (scripts), bats-core (tests), Homebrew (distribution), direnv (local workspace install)

# Shared setup/teardown for all bats test files.
# Load in each test file with:  load 'setup'

setup() {
  # Isolate from the real global git config and HOME
  export ORIG_HOME="$HOME"
  export HOME
  HOME="$(mktemp -d)"
  export XDG_CONFIG_HOME="$HOME/.config"

  # Minimal git identity (required for commits)
  git config --global user.name "Test User"
  git config --global user.email "test@example.com"
  git config --global init.defaultBranch main

  # Locate the git-cloak binary relative to tests/
  export GIT_CLOAK
  GIT_CLOAK="$(cd "$(dirname "$BATS_TEST_FILENAME")/.." && pwd)/bin/git-cloak"

  # Create a bare "remote" so push/pull tests work
  export TEST_REMOTE
  TEST_REMOTE="$(mktemp -d)/remote.git"
  git init --bare "$TEST_REMOTE"

  # Clone bare repo as our working copy
  export TEST_REPO
  TEST_REPO="$(mktemp -d)/myrepo"
  git clone "$TEST_REMOTE" "$TEST_REPO" 2>/dev/null

  # Initial tracked file + push to origin/main
  cd "$TEST_REPO"
  echo "# Test Repo" > README.md
  git add README.md
  git commit -m "Initial commit"
  git push -u origin main 2>/dev/null
}

teardown() {
  cd / 2>/dev/null || true
  rm -rf "$HOME" "$(dirname "$TEST_REMOTE")" "$(dirname "$TEST_REPO")"
  export HOME="$ORIG_HOME"
}
```

# lib/common.sh
# Path resolution and shared helpers. Sourced by bin/git-cloak before cmd-*.sh.
# Full implementation: Task 3.

_PROJECT_DIR="$(git rev-parse --show-toplevel 2>/dev/null)"

_WORKSPACE_DIR=""
_search_start="${_PROJECT_DIR:+$(dirname "$_PROJECT_DIR")}"
_search_start="${_search_start:-$PWD}"
_dir="$_search_start"
while [[ "$_dir" != "/" ]]; do
  if [[ -d "$_dir/.git-cloak" ]]; then
    _WORKSPACE_DIR="$_dir"
    break
  fi
  _dir="$(dirname "$_dir")"
done

_PROJECT_PERSONAL_DIR=""
[[ -n "$_PROJECT_DIR" ]] && \
  _PROJECT_PERSONAL_DIR="$(git -C "$_PROJECT_DIR" rev-parse --git-dir)/personal"

_WORKSPACE_CLOAK_DIR="${_WORKSPACE_DIR:+$_WORKSPACE_DIR/.git-cloak}"
_GLOBAL_CLOAK_DIR="${XDG_CONFIG_HOME:-$HOME/.config}/git-cloak"

_recompose_ignore() {
  [[ -d "$_PROJECT_PERSONAL_DIR" ]] || return 0
  local _composed="$_PROJECT_PERSONAL_DIR/ignore"
  : > "$_composed"
  if [[ -f "$_GLOBAL_CLOAK_DIR/ignore" ]]; then
    printf '# === global (source: %s) ===\n' "$_GLOBAL_CLOAK_DIR/ignore" >> "$_composed"
    cat "$_GLOBAL_CLOAK_DIR/ignore" >> "$_composed"
    printf '\n' >> "$_composed"
  fi
  if [[ -n "$_WORKSPACE_CLOAK_DIR" && -f "$_WORKSPACE_CLOAK_DIR/ignore" ]]; then
    printf '# === workspace (source: %s) ===\n' "$_WORKSPACE_CLOAK_DIR/ignore" >> "$_composed"
    cat "$_WORKSPACE_CLOAK_DIR/ignore" >> "$_composed"
    printf '\n' >> "$_composed"
  fi
  local _proj_src="$_PROJECT_PERSONAL_DIR/project-ignore"
  if [[ -f "$_proj_src" ]]; then
    printf '# === project ===\n' >> "$_composed"
    cat "$_proj_src" >> "$_composed"
    printf '\n' >> "$_composed"
  fi
}
```

```zsh
#!/usr/bin/env zsh
# git-cloak — personal file overlay manager for git
# Usage: git cloak <command> [args]
# git routes "git cloak <command>" here automatically when git-cloak is on PATH.

_cmd="${1:-}"
[[ -n "$_cmd" ]] && shift

# Resolve lib directory: Homebrew installs to lib/git-cloak/, local repo uses lib/
_SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
if [[ -d "$_SCRIPT_DIR/../lib/git-cloak" ]]; then
  _LIB="$(cd "$_SCRIPT_DIR/../lib/git-cloak" && pwd)"
else
  _LIB="$(cd "$_SCRIPT_DIR/../lib" && pwd)"
fi

if [[ ! -d "$_LIB" ]]; then
  echo "Error: git-cloak lib directory not found (expected at $_LIB)" >&2
  exit 1
fi

source "$_LIB/common.sh"

case "$_cmd" in
  init)        source "$_LIB/cmd-init.sh" ;;
  hide)        source "$_LIB/cmd-hide.sh" ;;
  watch)       source "$_LIB/cmd-watch.sh" ;;
  unwatch)     source "$_LIB/cmd-unwatch.sh" ;;
  list)        source "$_LIB/cmd-list.sh" ;;
  switch)      source "$_LIB/cmd-switch.sh" ;;
  refresh)     source "$_LIB/cmd-refresh.sh" ;;
  merge-main)  source "$_LIB/cmd-merge-main.sh" ;;
  sync)        source "$_LIB/cmd-sync.sh" ;;
  monitor)     source "$_LIB/cmd-monitor.sh" ;;
  "")
    echo "Usage: git cloak <command> [args]"
    echo "Commands: init, hide, watch, unwatch, list, switch, refresh, merge-main, sync, monitor"
    exit 1
    ;;
  *)
    echo "Unknown command: $_cmd"
    echo "Usage: git cloak <command> [args]"
    echo "Commands: init, hide, watch, unwatch, list, switch, refresh, merge-main, sync, monitor"
    exit 1
    ;;
esac
```

# lib/cmd-init.sh
# Usage: git cloak init [--workspace|--global]

_cloak_init() {
  case "${1:-}" in
    --workspace)
      if [[ -n "$_PROJECT_DIR" ]]; then
        echo "Error: you appear to be inside a git repo at '$_PROJECT_DIR'."
        echo "Run 'git cloak init --workspace' from outside a git repo (the workspace root)."
        exit 1
      fi
      mkdir -p ".git-cloak"
      touch ".git-cloak/ignore"
      echo "Workspace initialized at $(pwd)/.git-cloak"
      echo "Add patterns with: git cloak hide --workspace <pattern>"
      ;;
    --global)
      mkdir -p "$_GLOBAL_CLOAK_DIR"
      touch "$_GLOBAL_CLOAK_DIR/ignore"
      git config --global core.excludesFile "$_GLOBAL_CLOAK_DIR/ignore"
      echo "Global scope initialized at $_GLOBAL_CLOAK_DIR"
      echo "Add patterns with: git cloak hide --global <pattern>"
      ;;
    ""|--project)
      if [[ -z "$_PROJECT_DIR" ]]; then
        echo "Error: not inside a git repository."
        echo "Use --workspace or --global for non-repo init."
        exit 1
      fi
      mkdir -p "$_PROJECT_PERSONAL_DIR"
      touch "$_PROJECT_PERSONAL_DIR/project-ignore"
      git -C "$_PROJECT_DIR" config core.excludesFile "$_PROJECT_PERSONAL_DIR/ignore"
      _recompose_ignore
      echo "Project scope initialized."
      echo "  core.excludesFile → $_PROJECT_PERSONAL_DIR/ignore"
      if [[ -n "$_WORKSPACE_CLOAK_DIR" ]]; then
        echo "  Workspace patterns from: $_WORKSPACE_CLOAK_DIR/ignore"
      fi
      echo "Add patterns with: git cloak hide <pattern>"
      echo "Watch files with:  git cloak watch <file>"
      ;;
    *)
      echo "Usage: git cloak init [--workspace|--global]"
      exit 1
      ;;
  esac
}

_cloak_init "$@"
```

# lib/cmd-hide.sh
# Usage: git cloak hide [--project|--workspace|--global] <pattern>

_cloak_hide() {
  local _scope="auto"
  local _pattern=""

  case "${1:-}" in
    --project)   _scope="project";   _pattern="${2:-}" ;;
    --workspace) _scope="workspace"; _pattern="${2:-}" ;;
    --global)    _scope="global";    _pattern="${2:-}" ;;
    "")
      echo "Usage: git cloak hide [--project|--workspace|--global] <pattern>"
      exit 1
      ;;

      echo "$_pattern" >> "$_PROJECT_PERSONAL_DIR/project-ignore"

# lib/cmd-switch.sh
# Usage: git cloak switch <branch>
#        git cloak switch --restore-watch  (after manual conflict resolution)

_cloak_switch() {
  if [[ -z "$_PROJECT_DIR" ]]; then
    echo "Error: not inside a git repository."
    exit 1
  fi

  local _saved_watch="$_PROJECT_PERSONAL_DIR/switch-watched"
  local _watch_list="$_PROJECT_PERSONAL_DIR/watch-list"

  _is_monitored() {
    local _f="$1"
    [[ -f "$_watch_list" ]] && grep -v '^\s*#' "$_watch_list" | grep -qxF "$_f"
  }

  # --restore-watch: re-apply skip-worktree after manual conflict resolution
  if [[ "${1:-}" == "--restore-watch" ]]; then
    local _remaining
    _remaining=($(git -C "$_PROJECT_DIR" diff --name-only --diff-filter=U))
    if [[ ${#_remaining[@]} -gt 0 ]]; then
      echo "Unresolved conflicts remain:"
      for _f in "${_remaining[@]}"; do echo "  $_f"; done
      echo "Resolve them, stage with 'git add <file>', then run: git cloak switch --restore-watch"
      exit 1
    fi
    if [[ ! -f "$_saved_watch" ]]; then
      echo "Error: no saved watch list found at $_saved_watch"
      echo "Re-apply skip-worktree manually with: git cloak watch <file>"
      exit 1
    fi
    local _watched
    _watched=(${(f)"$(<$_saved_watch)"})
    echo "Unstaging all changes..."
    git -C "$_PROJECT_DIR" restore --staged . 2>/dev/null || true
    echo "Re-applying skip-worktree to ${#_watched[@]} files..."
    git -C "$_PROJECT_DIR" update-index --skip-worktree "${_watched[@]}"
    rm -f "$_saved_watch"
    echo "Done."
    return 0
  fi

  if [[ -z "${1:-}" ]]; then
    echo "Usage: git cloak switch <branch>"
    exit 1
  fi

  local _target="$1"

  # Verify the branch exists locally or on origin
  if ! git -C "$_PROJECT_DIR" rev-parse --verify "$_target" &>/dev/null && \
     ! git -C "$_PROJECT_DIR" rev-parse --verify "origin/$_target" &>/dev/null; then
    echo "Error: branch '$_target' not found locally or on origin."
    exit 1
  fi

  # Collect skip-worktree files
  local _watched
  _watched=($(git -C "$_PROJECT_DIR" ls-files -v | grep '^S' | awk '{print $2}'))

  if [[ ${#_watched[@]} -eq 0 ]]; then
    echo "No skip-worktree files found. Running plain checkout..."
    git -C "$_PROJECT_DIR" checkout "$_target"
    return $?
  fi

  # Save list for --restore-watch
  printf '%s\n' "${_watched[@]}" > "$_saved_watch"

  echo "Clearing skip-worktree on ${#_watched[@]} files..."
  git -C "$_PROJECT_DIR" update-index --no-skip-worktree "${_watched[@]}"

  echo "Stashing local changes..."
  git -C "$_PROJECT_DIR" stash push -m "git-cloak switch auto-stash ($_target)"

  echo "Switching to $_target..."
  git -C "$_PROJECT_DIR" checkout "$_target"
  local _checkout_exit=$?

  if [[ $_checkout_exit -ne 0 ]]; then
    echo "Checkout failed. Restoring stash and re-applying skip-worktree..."
    git -C "$_PROJECT_DIR" stash pop
    git -C "$_PROJECT_DIR" update-index --skip-worktree "${_watched[@]}"
    rm -f "$_saved_watch"
    exit $_checkout_exit
  fi

  echo "Restoring local changes..."
  git -C "$_PROJECT_DIR" stash pop
  local _stash_exit=$?

  if [[ $_stash_exit -ne 0 ]]; then
    local _conflicted
    _conflicted=($(git -C "$_PROJECT_DIR" diff --name-only --diff-filter=U))
    local _monitored_conflicts=()

    for _file in "${_conflicted[@]}"; do
      if _is_monitored "$_file"; then
        _monitored_conflicts+=("$_file")
        echo "MONITOR: $_file has conflicts — needs manual review"
      else
        git -C "$_PROJECT_DIR" checkout --theirs "$_file"
        git -C "$_PROJECT_DIR" add "$_file"
        echo "Auto-resolved (kept local): $_file"
      fi
    done

    if [[ ${#_monitored_conflicts[@]} -gt 0 ]]; then
      echo ""
      echo "Files needing manual resolution (in monitor list):"
      for _f in "${_monitored_conflicts[@]}"; do
        echo "  $_f"
        git -C "$_PROJECT_DIR" diff "$_f" || true
      done
      echo ""
      echo "Resolve each file, then: git add <file> && git cloak switch --restore-watch"
      exit 1
    fi
  fi

  echo "Unstaging all changes..."
  git -C "$_PROJECT_DIR" restore --staged . 2>/dev/null || true
  echo "Re-applying skip-worktree..."
  git -C "$_PROJECT_DIR" update-index --skip-worktree "${_watched[@]}"
  rm -f "$_saved_watch"
  echo ""
  echo "Done. Now on $(git -C "$_PROJECT_DIR" branch --show-current). Local changes preserved, skip-worktree re-applied."
}

_cloak_switch "$@"

_cloak_merge_main() {
  if [[ -z "$_PROJECT_DIR" ]]; then
    echo "Error: not inside a git repository."
    exit 1
  fi

  local _saved_watch="$_PROJECT_PERSONAL_DIR/merge-main-watched"
  local _watch_list="$_PROJECT_PERSONAL_DIR/watch-list"

  _is_monitored() {
    local _f="$1"
    [[ -f "$_watch_list" ]] && grep -v '^\s*#' "$_watch_list" | grep -qxF "$_f"
  }

  _handle_conflicts() {
    local _context="$1"  # "merge" or "stash"
    local _conflicted
    _conflicted=($(git -C "$_PROJECT_DIR" diff --name-only --diff-filter=U))
    local _monitored_conflicts=()

    for _file in "${_conflicted[@]}"; do
      if _is_monitored "$_file"; then
        _monitored_conflicts+=("$_file")
        echo "MONITOR: $_file has conflicts ($_context) — needs manual review"
      else
        git -C "$_PROJECT_DIR" checkout --theirs "$_file"
        git -C "$_PROJECT_DIR" add "$_file"
        echo "Auto-resolved (kept local): $_file"
      fi
    done

    if [[ ${#_monitored_conflicts[@]} -gt 0 ]]; then
      echo ""
      echo "Files needing manual resolution (in monitor list):"
      for _f in "${_monitored_conflicts[@]}"; do
        echo "  $_f"
        git -C "$_PROJECT_DIR" diff "$_f" || true
      done
      echo ""
      echo "Resolve each file, then: git add <file> && git cloak merge-main --restore-watch"
      return 1
    fi
    return 0
  }

  if [[ "${1:-}" == "--restore-watch" ]]; then
    local _remaining
    _remaining=($(git -C "$_PROJECT_DIR" diff --name-only --diff-filter=U))
    if [[ ${#_remaining[@]} -gt 0 ]]; then
      echo "Unresolved conflicts remain:"
      for _f in "${_remaining[@]}"; do echo "  $_f"; done
      echo "Resolve them, stage with 'git add <file>', then run: git cloak merge-main --restore-watch"
      exit 1
    fi
    if [[ ! -f "$_saved_watch" ]]; then
      echo "Error: no saved watch list found at $_saved_watch"
      echo "Re-apply skip-worktree manually with: git cloak watch <file>"
      exit 1
    fi
    local _watched
    _watched=(${(f)"$(<$_saved_watch)"})
    echo "Unstaging all changes..."
    git -C "$_PROJECT_DIR" restore --staged . 2>/dev/null || true
    echo "Re-applying skip-worktree to ${#_watched[@]} files..."
    git -C "$_PROJECT_DIR" update-index --skip-worktree "${_watched[@]}"
    rm -f "$_saved_watch"
    echo "Done."
    return 0
  fi

  local _current_branch
  _current_branch=$(git -C "$_PROJECT_DIR" branch --show-current)
  if [[ "$_current_branch" == "main" ]]; then
    echo "Already on main — use 'git cloak refresh' to pull instead."
    exit 1
  fi

  echo "Fetching origin/main..."
  git -C "$_PROJECT_DIR" fetch origin main

  local _watched
  _watched=($(git -C "$_PROJECT_DIR" ls-files -v | grep '^S' | awk '{print $2}'))

  if [[ ${#_watched[@]} -eq 0 ]]; then
    echo "No skip-worktree files found. Running plain merge..."
    git -C "$_PROJECT_DIR" merge origin/main
    return $?
  fi

  printf '%s\n' "${_watched[@]}" > "$_saved_watch"

  echo "Clearing skip-worktree on ${#_watched[@]} files..."
  git -C "$_PROJECT_DIR" update-index --no-skip-worktree "${_watched[@]}"

  echo "Stashing local changes..."
  git -C "$_PROJECT_DIR" stash push -m "git-cloak merge-main auto-stash"
  local _stash_was_created=$?

  echo "Merging origin/main into $_current_branch..."
  git -C "$_PROJECT_DIR" merge origin/main
  local _merge_exit=$?

  if [[ $_merge_exit -ne 0 ]]; then
    _handle_conflicts "merge" || exit 1
    # All merge conflicts auto-resolved — finalise merge commit
    git -C "$_PROJECT_DIR" commit --no-edit
  fi

  if [[ $_stash_was_created -eq 0 ]]; then
    echo "Restoring local changes..."
    git -C "$_PROJECT_DIR" stash pop
    local _stash_exit=$?

    if [[ $_stash_exit -ne 0 ]]; then
      _handle_conflicts "stash" || exit 1
    fi
  fi

  echo "Unstaging all changes..."
  git -C "$_PROJECT_DIR" restore --staged . 2>/dev/null || true
  echo "Re-applying skip-worktree..."
  git -C "$_PROJECT_DIR" update-index --skip-worktree "${_watched[@]}"
  rm -f "$_saved_watch"
  echo ""
  echo "Done. $_current_branch is now up to date with origin/main."
}

_cloak_merge_main "$@"
```

# lib/cmd-sync.sh
# Usage: git cloak sync <file-path-relative-to-repo-root>
# Fetches the upstream version of a skip-worktree file for manual merge review.

_cloak_sync() {
  if [[ -z "${1:-}" ]]; then
    echo "Usage: git cloak sync <file-path-relative-to-repo-root>"
    exit 1
  fi
  if [[ -z "$_PROJECT_DIR" ]]; then
    echo "Error: not inside a git repository."
    exit 1
  fi

  local _file="$1"
  local _full_path="$_PROJECT_DIR/$_file"
  local _backup_path="${_full_path}.local-backup"

  if [[ ! -f "$_full_path" ]]; then
    echo "Error: $_full_path does not exist."
    exit 1
  fi

  if ! git -C "$_PROJECT_DIR" ls-files -v "$_file" | grep -q '^S'; then
    echo "Warning: $_file is not currently skip-worktree — syncing anyway."
  fi

  echo "Backing up your local version to ${_file}.local-backup..."
  cp "$_full_path" "$_backup_path"

  echo "Clearing skip-worktree and fetching upstream..."
  git -C "$_PROJECT_DIR" update-index --no-skip-worktree "$_file"
  git -C "$_PROJECT_DIR" fetch origin

  echo "Applying upstream version (origin/main)..."
  git -C "$_PROJECT_DIR" checkout origin/main -- "$_file"

  echo "Re-applying skip-worktree..."
  git -C "$_PROJECT_DIR" update-index --skip-worktree "$_file"

  echo ""
  echo "Done. Upstream version is now in place."
  echo "Your local customizations are saved at: ${_file}.local-backup"
  echo ""
  echo "Diff (upstream vs your local backup):"
  diff --color=always "$_full_path" "$_backup_path" || true
  echo ""
  echo "When done merging your customizations back: rm ${_file}.local-backup"
}

_cloak_sync "$@"

# lib/cmd-monitor.sh
# Usage: git cloak monitor <add|remove|list> [file]
# Manages the per-project conflict-pause watch list.

_cloak_monitor() {
  if [[ -z "$_PROJECT_DIR" ]]; then
    echo "Error: not inside a git repository."
    exit 1
  fi

  local _watch_list="$_PROJECT_PERSONAL_DIR/watch-list"

  case "${1:-}" in
    add)
      if [[ -z "${2:-}" ]]; then
        echo "Usage: git cloak monitor add <file-path>"
        exit 1
      fi
      if [[ -f "$_watch_list" ]] && grep -qxF "$2" "$_watch_list"; then
        echo "Already in monitor list: $2"
      else
        mkdir -p "$_PROJECT_PERSONAL_DIR"
        echo "$2" >> "$_watch_list"
        echo "Added to monitor list: $2"
      fi
      ;;
    remove)
      if [[ -z "${2:-}" ]]; then
        echo "Usage: git cloak monitor remove <file-path>"
        exit 1
      fi
      if [[ ! -f "$_watch_list" ]] || ! grep -qxF "$2" "$_watch_list"; then
        echo "Not in monitor list: $2"
      else
        # macOS-compatible in-place sed
        local _escaped
        _escaped="$(printf '%s\n' "$2" | sed 's/[[\.*^$()+?{|]/\\&/g')"
        sed -i '' "/^${_escaped}$/d" "$_watch_list"
        echo "Removed from monitor list: $2"
      fi
      ;;
    list)
      if [[ ! -f "$_watch_list" ]] || ! grep -qv '^\s*#' "$_watch_list" 2>/dev/null; then
        echo "Monitor list is empty."
      else
        echo "=== conflict monitor list ==="
        grep -v '^\s*#' "$_watch_list" | grep -v '^$' | awk '{print "  " $0}'
      fi
      ;;
    "")
      echo "Usage: git cloak monitor <add|remove|list> [file]"
      exit 1
      ;;
    *)
      echo "Unknown subcommand: $1"
      echo "Usage: git cloak monitor <add|remove|list> [file]"
      exit 1
      ;;
  esac
}

_cloak_monitor "$@"

# git-cloak

Personal file overlay manager for git. Hide local changes to tracked files and personal untracked files safely across branch switches, pulls, and merges.

## What it does

Two hiding mechanisms:
- **skip-worktree** — git stops seeing your local changes to a tracked file
- **Scoped ignore** — personal untracked files that git never sees

Three config scopes:
- **global** — `~/.config/git-cloak/ignore` — applies to all repos on the machine
- **workspace** — `<workspace>/.git-cloak/ignore` — all repos under a workspace directory
- **project** — `.git/personal/project-ignore` — single repo only

## Installation

### Homebrew (recommended)
```bash
brew tap <user>/tap
brew install git-cloak
```

### Local workspace (direnv)
Clone into a workspace directory and add to `.envrc`:
```bash
PATH_add /path/to/git-cloak/bin
```

## Quick start

```bash
# Initialize global scope (optional — affects all repos)
git cloak init --global

# Initialize a workspace root (run from the directory above your repos)
cd ~/Developer/myworkspace
git cloak init --workspace

# Initialize a repo
cd ~/Developer/myworkspace/myrepo
git cloak init

# Hide a personal untracked file
git cloak hide ".env.local"
git cloak hide --workspace "*.local-backup"
git cloak hide --global ".DS_Store"

# Watch a tracked file (freeze your local changes)
git cloak watch config/database.yml

# Show all watched files and active ignore patterns
git cloak list
```

## Safe branch operations

```bash
# Switch branches safely (preserves skip-worktree files)
git cloak switch <branch>

# Pull safely
git cloak refresh

# Merge origin/main into current branch safely
git cloak merge-main
```

If a monitored file conflicts, the command pauses for manual resolution. After resolving:

## Migration from existing setup

If you have existing skip-worktree files and a scoped gitignore:

1. Install git-cloak (Homebrew or local)
2. Run `git cloak init --workspace` from your workspace root
3. Run `git cloak init` from each repo — existing skip-worktree state is preserved
4. Migrate old gitignore patterns: `git cloak hide --workspace "<pattern>"`
